import React from "react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { Button } from "@/components/ui/button";
import { Building2, MapPin, DollarSign, Clock, Briefcase } from "lucide-react";
import { motion } from "framer-motion";
import { format } from "date-fns";
import { ru } from "date-fns/locale";

const employmentTypes = {
  full_time: "Полная занятость",
  part_time: "Частичная занятость",
  remote: "Удаленная работа",
  hybrid: "Гибрид"
};

const experienceLevels = {
  no_experience: "Без опыта",
  "1_3_years": "1-3 года",
  "3_6_years": "3-6 лет",
  "6_plus_years": "6+ лет"
};

export default function VacancyCard({ vacancy, onApply }) {
  return (
    <motion.div
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      whileHover={{ y: -4 }}
      transition={{ duration: 0.2 }}
    >
      <Card className="h-full hover:shadow-2xl transition-all duration-300 border-0 shadow-lg overflow-hidden group">
        <div className="absolute top-0 left-0 right-0 h-1 bg-gradient-to-r from-indigo-600 to-cyan-600" />
        <CardHeader className="pb-4">
          <div className="flex items-start justify-between mb-3">
            <div className="flex items-center gap-3">
              <div className="w-12 h-12 bg-gradient-to-br from-indigo-100 to-cyan-100 rounded-xl flex items-center justify-center group-hover:scale-110 transition-transform">
                <Briefcase className="w-6 h-6 text-indigo-600" />
              </div>
              <div>
                <CardTitle className="text-xl font-bold text-gray-900 leading-tight">
                  {vacancy.position}
                </CardTitle>
                <div className="flex items-center gap-1 text-sm text-gray-600 mt-1">
                  <Building2 className="w-4 h-4" />
                  <span>{vacancy.company_name}</span>
                </div>
              </div>
            </div>
          </div>

          <div className="flex flex-wrap gap-2">
            {vacancy.employment_type && (
              <Badge variant="secondary" className="bg-indigo-50 text-indigo-700 border-indigo-200">
                {employmentTypes[vacancy.employment_type]}
              </Badge>
            )}
            {vacancy.experience && (
              <Badge variant="secondary" className="bg-cyan-50 text-cyan-700 border-cyan-200">
                {experienceLevels[vacancy.experience]}
              </Badge>
            )}
          </div>
        </CardHeader>

        <CardContent className="space-y-4">
          <p className="text-gray-600 line-clamp-3 text-sm leading-relaxed">
            {vacancy.description}
          </p>

          <div className="space-y-2 text-sm">
            {vacancy.location && (
              <div className="flex items-center gap-2 text-gray-600">
                <MapPin className="w-4 h-4 text-gray-400" />
                <span>{vacancy.location}</span>
              </div>
            )}
            {(vacancy.salary_from || vacancy.salary_to) && (
              <div className="flex items-center gap-2 text-gray-600">
                <DollarSign className="w-4 h-4 text-gray-400" />
                <span className="font-semibold text-green-600">
                  {vacancy.salary_from && vacancy.salary_to
                    ? `${vacancy.salary_from.toLocaleString()} - ${vacancy.salary_to.toLocaleString()} ₽`
                    : vacancy.salary_from
                    ? `от ${vacancy.salary_from.toLocaleString()} ₽`
                    : `до ${vacancy.salary_to.toLocaleString()} ₽`}
                </span>
              </div>
            )}
            {vacancy.published_date && (
              <div className="flex items-center gap-2 text-gray-500 text-xs">
                <Clock className="w-3 h-3" />
                <span>
                  Опубликовано: {format(new Date(vacancy.published_date), 'd MMMM yyyy', { locale: ru })}
                </span>
              </div>
            )}
          </div>

          <Button
            onClick={() => onApply(vacancy)}
            className="w-full bg-gradient-to-r from-indigo-600 to-cyan-600 hover:from-indigo-700 hover:to-cyan-700 text-white shadow-lg"
          >
            Откликнуться
          </Button>
        </CardContent>
      </Card>
    </motion.div>
  );
}
