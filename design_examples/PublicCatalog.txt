import React, { useState } from "react";
import { useQuery } from "@tanstack/react-query";
import { base44 } from "@/api/base44Client";
import { Link } from "react-router-dom";
import { createPageUrl } from "@/utils";
import { 
  Briefcase, GraduationCap, MapPin, DollarSign, Clock, 
  Building2, Search, Filter, Calendar, TrendingUp
} from "lucide-react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs";
import { Badge } from "@/components/ui/badge";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Skeleton } from "@/components/ui/skeleton";
import { format } from "date-fns";
import { ru } from "date-fns/locale";

import VacancyCard from "../components/catalog/VacancyCard";
import InternshipCard from "../components/catalog/InternshipCard";
import ApplyModal from "../components/catalog/ApplyModal";
import StatsSection from "../components/catalog/StatsSection";

export default function PublicCatalog() {
  const [searchQuery, setSearchQuery] = useState("");
  const [selectedVacancy, setSelectedVacancy] = useState(null);
  const [selectedInternship, setSelectedInternship] = useState(null);
  const [showApplyModal, setShowApplyModal] = useState(false);

  const { data: vacancies, isLoading: vacanciesLoading } = useQuery({
    queryKey: ['vacancies', 'published'],
    queryFn: () => base44.entities.Vacancy.filter({ status: 'published' }, '-published_date'),
    initialData: [],
  });

  const { data: internships, isLoading: internshipsLoading } = useQuery({
    queryKey: ['internships', 'published'],
    queryFn: () => base44.entities.Internship.filter({ status: 'published' }, '-published_date'),
    initialData: [],
  });

  const filteredVacancies = vacancies.filter(v =>
    v.position?.toLowerCase().includes(searchQuery.toLowerCase()) ||
    v.company_name?.toLowerCase().includes(searchQuery.toLowerCase()) ||
    v.description?.toLowerCase().includes(searchQuery.toLowerCase())
  );

  const filteredInternships = internships.filter(i =>
    i.specialty?.toLowerCase().includes(searchQuery.toLowerCase()) ||
    i.university_name?.toLowerCase().includes(searchQuery.toLowerCase()) ||
    i.description?.toLowerCase().includes(searchQuery.toLowerCase())
  );

  const handleApply = (vacancy) => {
    setSelectedVacancy(vacancy);
    setSelectedInternship(null);
    setShowApplyModal(true);
  };

  const handleInternshipContact = (internship) => {
    setSelectedInternship(internship);
    setSelectedVacancy(null);
    setShowApplyModal(true);
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-slate-50 via-blue-50 to-indigo-50">
      {/* Hero Section */}
      <section className="relative overflow-hidden bg-gradient-to-r from-indigo-600 via-blue-600 to-cyan-600 text-white">
        <div className="absolute inset-0 bg-grid-white/[0.05] bg-[size:20px_20px]" />
        <div className="relative max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-24">
          <div className="text-center max-w-3xl mx-auto">
            <h1 className="text-5xl md:text-6xl font-bold mb-6 leading-tight">
              Карьера в технологическом
              <span className="block bg-gradient-to-r from-cyan-300 to-blue-300 bg-clip-text text-transparent">
                кластере Москвы
              </span>
            </h1>
            <p className="text-xl text-blue-100 mb-8">
              Платформа для поиска работы и стажировок в ОЭЗ «Технополис Москва»
            </p>
            <div className="flex flex-col sm:flex-row gap-4 justify-center">
              <Button 
                size="lg" 
                className="bg-white text-indigo-600 hover:bg-blue-50 shadow-xl"
                onClick={() => document.getElementById('catalog')?.scrollIntoView({ behavior: 'smooth' })}
              >
                <Briefcase className="w-5 h-5 mr-2" />
                Найти вакансию
              </Button>
              <Button 
                size="lg" 
                variant="outline"
                className="border-2 border-white text-white hover:bg-white/10"
                onClick={() => base44.auth.redirectToLogin(createPageUrl("AdminDashboard"))}
              >
                <GraduationCap className="w-5 h-5 mr-2" />
                Разместить стажировку
              </Button>
            </div>
          </div>
        </div>
        <div className="absolute bottom-0 left-0 right-0">
          <svg viewBox="0 0 1440 120" fill="none" xmlns="http://www.w3.org/2000/svg">
            <path d="M0 120L60 105C120 90 240 60 360 45C480 30 600 30 720 37.5C840 45 960 60 1080 67.5C1200 75 1320 75 1380 75L1440 75V120H1380C1320 120 1200 120 1080 120C960 120 840 120 720 120C600 120 480 120 360 120C240 120 120 120 60 120H0Z" fill="rgb(248 250 252)"/>
          </svg>
        </div>
      </section>

      {/* Stats Section */}
      <StatsSection 
        vacanciesCount={vacancies.length}
        internshipsCount={internships.length}
      />

      {/* Catalog Section */}
      <section id="catalog" className="py-16">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
          {/* Search Bar */}
          <div className="mb-12">
            <div className="relative max-w-2xl mx-auto">
              <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-gray-400" />
              <Input
                type="text"
                placeholder="Поиск по должности, компании, специальности..."
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                className="pl-12 pr-4 h-14 text-lg border-2 border-gray-200 focus:border-indigo-500 rounded-2xl shadow-lg"
              />
            </div>
          </div>

          {/* Tabs */}
          <Tabs defaultValue="vacancies" className="w-full">
            <TabsList className="grid w-full max-w-md mx-auto grid-cols-2 mb-8 bg-white p-1 rounded-2xl shadow-md">
              <TabsTrigger 
                value="vacancies"
                className="rounded-xl data-[state=active]:bg-gradient-to-r data-[state=active]:from-indigo-600 data-[state=active]:to-blue-600 data-[state=active]:text-white"
              >
                <Briefcase className="w-4 h-4 mr-2" />
                Вакансии ({filteredVacancies.length})
              </TabsTrigger>
              <TabsTrigger 
                value="internships"
                className="rounded-xl data-[state=active]:bg-gradient-to-r data-[state=active]:from-purple-600 data-[state=active]:to-pink-600 data-[state=active]:text-white"
              >
                <GraduationCap className="w-4 h-4 mr-2" />
                Стажировки ({filteredInternships.length})
              </TabsTrigger>
            </TabsList>

            <TabsContent value="vacancies">
              {vacanciesLoading ? (
                <div className="grid md:grid-cols-2 lg:grid-cols-3 gap-6">
                  {Array(6).fill(0).map((_, i) => (
                    <Card key={i}>
                      <CardHeader>
                        <Skeleton className="h-6 w-3/4 mb-2" />
                        <Skeleton className="h-4 w-1/2" />
                      </CardHeader>
                      <CardContent>
                        <Skeleton className="h-20 w-full" />
                      </CardContent>
                    </Card>
                  ))}
                </div>
              ) : filteredVacancies.length === 0 ? (
                <div className="text-center py-16">
                  <Briefcase className="w-16 h-16 text-gray-300 mx-auto mb-4" />
                  <h3 className="text-xl font-semibold text-gray-600 mb-2">
                    Вакансии не найдены
                  </h3>
                  <p className="text-gray-500">
                    Попробуйте изменить параметры поиска
                  </p>
                </div>
              ) : (
                <div className="grid md:grid-cols-2 lg:grid-cols-3 gap-6">
                  {filteredVacancies.map((vacancy) => (
                    <VacancyCard
                      key={vacancy.id}
                      vacancy={vacancy}
                      onApply={handleApply}
                    />
                  ))}
                </div>
              )}
            </TabsContent>

            <TabsContent value="internships">
              {internshipsLoading ? (
                <div className="grid md:grid-cols-2 lg:grid-cols-3 gap-6">
                  {Array(6).fill(0).map((_, i) => (
                    <Card key={i}>
                      <CardHeader>
                        <Skeleton className="h-6 w-3/4 mb-2" />
                        <Skeleton className="h-4 w-1/2" />
                      </CardHeader>
                      <CardContent>
                        <Skeleton className="h-20 w-full" />
                      </CardContent>
                    </Card>
                  ))}
                </div>
              ) : filteredInternships.length === 0 ? (
                <div className="text-center py-16">
                  <GraduationCap className="w-16 h-16 text-gray-300 mx-auto mb-4" />
                  <h3 className="text-xl font-semibold text-gray-600 mb-2">
                    Стажировки не найдены
                  </h3>
                  <p className="text-gray-500">
                    Попробуйте изменить параметры поиска
                  </p>
                </div>
              ) : (
                <div className="grid md:grid-cols-2 lg:grid-cols-3 gap-6">
                  {filteredInternships.map((internship) => (
                    <InternshipCard
                      key={internship.id}
                      internship={internship}
                      onContact={handleInternshipContact}
                    />
                  ))}
                </div>
              )}
            </TabsContent>
          </Tabs>
        </div>
      </section>

      {/* Apply Modal */}
      {showApplyModal && (
        <ApplyModal
          vacancy={selectedVacancy}
          internship={selectedInternship}
          onClose={() => {
            setShowApplyModal(false);
            setSelectedVacancy(null);
            setSelectedInternship(null);
          }}
        />
      )}
    </div>
  );
}
